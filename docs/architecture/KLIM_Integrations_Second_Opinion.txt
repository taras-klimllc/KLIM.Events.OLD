```csharp
using Azure.Identity;
using Azure.Core;
using Microsoft.Data.SqlClient;

namespace KLIM.Integrations.Affinity.Infrastructure;

public sealed class SqlAuthenticationService
{
    private const string AZURE_SQL_SCOPE = "https://database.windows.net/.default";

    public async Task<string> AcquireTokenAsync(CancellationToken cancellationToken = default)
    {
        var credential = new DefaultAzureCredential();
        var token = await credential.GetTokenAsync(
            new TokenRequestContext(new[] { AZURE_SQL_SCOPE }), 
            cancellationToken);
        return token.Token;
    }

    public string SanitizeConnectionString(string connectionString, bool useAzureAd)
    {
        if (!useAzureAd) return connectionString;
        var builder = new SqlConnectionStringBuilder(connectionString);
        builder.Remove("User ID");
        builder.Remove("Password");
        return builder.ConnectionString;
    }
}
```

**PROBLEM 2: Incorrect Core Table Modification**
The proposed solution suggests modifying dbo.Issuers:
```sql
-- WRONG - Cannot modify existing core table
ALTER TABLE dbo.Issuers ADD 
    AffinityExternalId NVARCHAR(128) NULL,

```

**ANALYSIS**: This violates separation of concerns and creates tight coupling between
core business entities and integration systems. Future integrations would require
further table modifications, creating a maintenance nightmare.

**CORRECT APPROACH**: Create integration mapping tables that reference core entities:

```csharp
        var root = doc.RootElement;

        string affinityOrgId = root.GetPropertyOrDefault("organizationId")
```

```csharp
// src/KLIM.Integrations.Affinity.SqlWriter/Infrastructure/IntegrationDiagnosticService.cs - NEW
using Microsoft.Extensions.Options;
using Microsoft.Data.SqlClient;
using KLIM.Integrations.Affinity.SqlWriter.Configuration;

namespace KLIM.Integrations.Affinity.SqlWriter.Infrastructure;

/// <summary>
/// Diagnostic service for integration status monitoring (follows KLIM.Events pattern)
/// </summary>
public sealed class IntegrationDiagnosticService : BackgroundService
{
    private readonly ILogger<IntegrationDiagnosticService> _logger;
    private readonly SqlAuthenticationService _authService;
    private readonly IntegrationOptions _config;
    private readonly WebhookOptions _fallbackConfig;

    public IntegrationDiagnosticService(
        ILogger<IntegrationDiagnosticService> logger,
        SqlAuthenticationService authService,
        IOptions<IntegrationOptions> config,
        IOptions<WebhookOptions> fallbackConfig)
    {
        _logger = logger;
        _authService = authService;
        _config = config.Value;
        _fallbackConfig = fallbackConfig.Value;
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        // Run diagnostics once on startup, then every configured interval
        await RunDiagnosticsAsync();
        
        using var timer = new PeriodicTimer(TimeSpan.FromMinutes(_config.DiagnosticsIntervalMinutes));
        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                await timer.WaitForNextTickAsync(stoppingToken);
                await RunDiagnosticsAsync();
            }
            catch (OperationCanceledException) when (stoppingToken.IsCancellationRequested)
            {
                break;
            }
        }
    }

    private async Task RunDiagnosticsAsync()
    {
        try
        {
            var (connectionString, useAzureAd) = ResolveConnectionConfig();
            if (string.IsNullOrWhiteSpace(connectionString))
            {
                _logger.LogWarning("?? INTEGRATION DIAGNOSTICS: No connection string available");
                return;
            }

            connectionString = _authService.SanitizeConnectionString(connectionString, useAzureAd);

            await using var connection = new SqlConnection(connectionString);
            if (useAzureAd)
            {
                connection.AccessToken = await _authService.AcquireTokenAsync();
            }
            await connection.OpenAsync();

            // Check webhook events
            const string webhookSql = @"
            SELECT 
                COUNT(*) as TotalEvents,
                COUNT(DISTINCT EventType) as UniqueEventTypes,
                MAX(ReceivedAt) as LastReceived
            FROM aff.WebhookEvents WITH (NOLOCK)";

            var webhookStats = await connection.QuerySingleOrDefaultAsync(webhookSql);

            // Check integration mappings
            const string mappingSql = @"
            SELECT 
                COUNT(*) as TotalMappings,
                COUNT(CASE WHEN SyncStatus = 'Active' THEN 1 END) as ActiveMappings,
                COUNT(CASE WHEN SyncStatus = 'Merged' THEN 1 END) as MergedMappings,
                MAX(UpdatedAtUtc) as LastUpdate
            FROM integration.AffinityOrganizations WITH (NOLOCK)";

            var mappingStats = await connection.QuerySingleOrDefaultAsync(mappingSql);

            // Check core issuer references
            const string issuerSql = @"
            SELECT COUNT(DISTINCT KLIMIssuerID) as UniqueIssuersLinked
            FROM integration.AffinityOrganizations WITH (NOLOCK)
            WHERE KLIMIssuerID IS NOT NULL";

            var issuerStats = await connection.QuerySingleOrDefaultAsync<int?>(issuerSql) ?? 0;

            _logger.LogInformation(
                "?? INTEGRATION STATUS: Webhooks={TotalEvents} types={UniqueTypes} | Mappings={TotalMappings} (Active={ActiveMappings}, Merged={MergedMappings}) | LinkedIssuers={LinkedIssuers}",
                webhookStats?.TotalEvents ?? 0, 
                webhookStats?.UniqueEventTypes ?? 0,
                mappingStats?.TotalMappings ?? 0, 
                mappingStats?.ActiveMappings ?? 0, 
                mappingStats?.MergedMappings ?? 0,
                issuerStats);

            // Check for recent activity
            if (webhookStats?.LastReceived != null)
            {
                var lastReceived = (DateTime)webhookStats.LastReceived;
                var timeSinceLastWebhook = DateTime.UtcNow - lastReceived;
                
                if (timeSinceLastWebhook > TimeSpan.FromHours(1))
                {
                    _logger.LogWarning("??  No webhook activity for {Hours} hours (last: {LastReceived:yyyy-MM-dd HH:mm:ss} UTC)", 
                        (int)timeSinceLastWebhook.TotalHours, lastReceived);
                }
            }

            // Database health check
            _logger.LogInformation("?? Database connection healthy (AzureAD: {UseAzureAd})", useAzureAd);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "?? INTEGRATION DIAGNOSTICS FAILED");
        }
    }

    private (string ConnectionString, bool UseAzureAd) ResolveConnectionConfig()
    {
        // Follow KLIM.Events pattern: primary config, then fallback
        if (!string.IsNullOrWhiteSpace(_config.ConnectionString))
            return (_config.ConnectionString, _config.UseAzureAd);

        if (!string.IsNullOrWhiteSpace(_fallbackConfig.ConnectionString))
        {
            _logger.LogDebug("Using fallback connection from Webhooks config");
            return (_fallbackConfig.ConnectionString, _fallbackConfig.UseAzureAd);
        }

        return (string.Empty, false);
    }
}
```

```csharp
// Webhook endpoint with Azure AD support
app.MapPost("/webhooks/affinity/{secret}", async (
    HttpRequest request, 
    string secret, 
    IConfiguration config, 
    SqlAuthenticationService sqlAuth,
    IRabbitPublisher publisher, 
    ILoggerFactory lf, 
    CancellationToken ct) =>
{
    var logger = lf.CreateLogger("AffinityWebhook");
    var webhookOptions = config.GetSection("Webhooks").Get<WebhookOptions>()!;
    
    if (string.IsNullOrWhiteSpace(webhookOptions.Secret) || !string.Equals(secret, webhookOptions.Secret, StringComparison.Ordinal))
    {
        logger.LogWarning("Webhook secret mismatch for delivery {DeliveryKey}", 
            request.Headers["X-Affinity-Delivery"].FirstOrDefault() ?? "unknown");
        return Results.Unauthorized();
    }

    var eventType = request.Headers["X-Affinity-Event"].FirstOrDefault() ?? "unknown";
    var deliveryKey = request.Headers["X-Affinity-Delivery"].FirstOrDefault() ?? Guid.NewGuid().ToString("N");

    using var reader = new StreamReader(request.Body);
    var body = await reader.ReadToEndAsync(ct);
    if (string.IsNullOrWhiteSpace(body))
        return Results.BadRequest("Empty payload");

    // Persist raw webhook with Azure AD support
    var integrationOptions = config.GetSection("Integration").Get<IntegrationOptions>()!;
    var (connectionString, useAzureAd) = ResolveConnectionConfig(integrationOptions, webhookOptions);
    
    if (!string.IsNullOrWhiteSpace(connectionString))
    {
        connectionString = sqlAuth.SanitizeConnectionString(connectionString, useAzureAd);
        
        await using var conn = new SqlConnection(connectionString);
        if (useAzureAd)
        {
            conn.AccessToken = await sqlAuth.AcquireTokenAsync(ct);
        }
        await conn.OpenAsync(ct);
        
        var hash = SqlHelpers.ComputePayloadHash(eventType, body);
        await SqlHelpers.InsertWebhookEventIfNewAsync(conn, eventType, deliveryKey, hash, body, ct);
    }

    // Publish to aligned topic exchange
    var routingKey = BuildRoutingKey(eventType, body);
    var headers = new Dictionary<string, object>
    {
        ["delivery-key"] = deliveryKey,
        ["source"] = "affinity",
        ["correlation-id"] = Guid.NewGuid().ToString(),
        ["received-at"] = DateTimeOffset.UtcNow.ToString("O")
    };

    await publisher.PublishAsync(routingKey, System.Text.Encoding.UTF8.GetBytes(body), headers);

    logger.LogInformation("Processed Affinity webhook: {EventType} -> {RoutingKey} (delivery: {DeliveryKey})",
        eventType, routingKey, deliveryKey);

    return Results.Accepted();
});

app.Run();

// Helper method following KLIM.Events pattern
static (string ConnectionString, bool UseAzureAd) ResolveConnectionConfig(IntegrationOptions integration, WebhookOptions webhook)
{
    if (!string.IsNullOrWhiteSpace(integration.ConnectionString))
        return (integration.ConnectionString, integration.UseAzureAd);

    if (!string.IsNullOrWhiteSpace(webhook.ConnectionString))
        return (webhook.ConnectionString, webhook.UseAzureAd);

    return (string.Empty, false);
}

// Updated routing key generation
static string BuildRoutingKey(string eventTypeHeader, string body)
{
    var evt = (eventTypeHeader ?? string.Empty).Trim();
    if (string.IsNullOrEmpty(evt))
    {
        try
        {
            var node = JsonNode.Parse(body);
            evt = node?["type"]?.GetValue<string>() ?? "unknown";
        }
        catch { evt = "unknown"; }
    }
    
    return $"klim.integration.affinity.{evt.Replace(' ', '.').ToLowerInvariant()}.v1";
}
```
